cmake_minimum_required(VERSION 3.21)
project(GREMLIN VERSION 0.2.1 LANGUAGES CXX)
include(CTest)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(GREMLIN_SQLITE_RUNTIME "" CACHE FILEPATH "SQLite DLL for local Windows executables")
if(WIN32 AND GREMLIN_SQLITE_RUNTIME)
    configure_file("${GREMLIN_SQLITE_RUNTIME}" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/sqlite3.dll" COPYONLY)
endif()

function(gremlin_target target)
    target_compile_features(${target} PRIVATE cxx_std_20)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()
